### 0-1. `src/api/axios.ts` (그대로 교체)

```ts
import axios, { AxiosError, InternalAxiosRequestConfig } from 'axios';
import { tokenStore } from '../services/tokenStore';

export const BASE_URL = 'http://10.0.2.2:8080'; // 환경에 맞게

const api = axios.create({ baseURL: BASE_URL, timeout: 15000 });

api.interceptors.request.use(config => {
  const token = tokenStore.accessToken;
  if (token) config.headers.Authorization = `Bearer ${token}`;
  return config;
});

let refreshing: Promise<string> | null = null; // 동시에 401 이 여러 개 와도 갱신은 한 번만

async function refreshAccessToken(): Promise<string> {
  const { data } = await axios.post(`${BASE_URL}/auth/refresh`, { refreshToken: tokenStore.refreshToken });
  await tokenStore.save(data.accessToken, data.refreshToken); // refreshToken 은 1회용: 새 값으로 반드시 덮어쓰기
  return data.accessToken;
}

api.interceptors.response.use(
  res => res,
  async (error: AxiosError) => {
    const original = error.config as InternalAxiosRequestConfig & { _retried?: boolean };
    const isAuthCall = original?.url?.startsWith('/auth/');
    if (error.response?.status !== 401 || !original || original._retried || isAuthCall) {
      return Promise.reject(error);
    }
    original._retried = true;
    try {
      refreshing = refreshing ?? refreshAccessToken().finally(() => { refreshing = null; });
      original.headers.Authorization = `Bearer ${await refreshing}`;
      return api(original);
    } catch (e) {
      await tokenStore.clear();
      onSessionExpired?.(); // App 에서 로그인 화면으로 보내는 콜백 (아래 1-2)
      return Promise.reject(e);
    }
  },
);

let onSessionExpired: (() => void) | undefined;
export const setSessionExpiredHandler = (fn: () => void) => { onSessionExpired = fn; };

/** 서버가 준 한국어 메시지를 꺼낸다 (토스트에 그대로 쓰면 됨) */
export function errorMessage(e: unknown, fallback = '잠시 후 다시 시도해 주세요.') {
  const data = (e as AxiosError<{ message?: string }>)?.response?.data;
  return data?.message ?? fallback;
}

export default api;
```

### 0-2. `src/services/tokenStore.ts`

```ts
import AsyncStorage from '@react-native-async-storage/async-storage';

const KEY = 'pinkok.tokens';

export const tokenStore = {
  accessToken: null as string | null,
  refreshToken: null as string | null,

  async load() {
    const raw = await AsyncStorage.getItem(KEY);
    if (raw) {
      const t = JSON.parse(raw);
      this.accessToken = t.accessToken;
      this.refreshToken = t.refreshToken;
    }
    return !!this.refreshToken;
  },
  async save(accessToken: string, refreshToken: string) {
    this.accessToken = accessToken;
    this.refreshToken = refreshToken;
    await AsyncStorage.setItem(KEY, JSON.stringify({ accessToken, refreshToken }));
  },
  async clear() {
    this.accessToken = null;
    this.refreshToken = null;
    await AsyncStorage.removeItem(KEY);
  },
};
```

### 0-3. AI 요청 공통 — "요청 → 상태 반복 조회"

AI 추출·추천은 **즉시 `202 PENDING`** 으로 응답하고 백그라운드에서 처리합니다(10~30초). 결과가 `SUCCESS`/`FAILED`가 될 때까지 조회합니다. 화면을 나가면 멈춰야 하므로 취소 신호를 받습니다.

```ts
// src/api/aiApi.ts
import api from './axios';

export type AiRequest = {
  id: number; status: 'PENDING' | 'PROCESSING' | 'SUCCESS' | 'FAILED';
  errorCode: string | null; title: string | null;
  places: { name: string; address: string; category: string; description: string }[] | null;
};

export async function waitForAi(id: number, signal: { cancelled: boolean }, maxSeconds = 90): Promise<AiRequest | null> {
  for (let i = 0; i < maxSeconds / 2.5; i++) {
    await new Promise(r => setTimeout(r, 2500));
    if (signal.cancelled) return null;
    const { data } = await api.get<AiRequest>(`/ai-requests/${id}`);
    if (data.status === 'SUCCESS' || data.status === 'FAILED') return data;
  }
  return null; // 시간 초과 → "오래 걸리고 있어요" 안내
}
```

### 0-4. ID 타입 주의

`PlaceItem`으로 바꾸는 함수 (서버는 **일자·방문 순서대로 정렬**해서 줍니다. `dayId`가 없는 핀은 "날짜 미배정"):

```ts
const TRANSIT_LABEL = { WALK: '도보', CAR: '자동차', BUS: '버스', TRAIN: '기차' } as const;
const TRANSIT_TYPE = { WALK: 'walk', CAR: 'car', BUS: 'bus', TRAIN: 'bus' } as const;

export function toPlaceItems(items: ItemDto[]): PlaceItem[] {
  return items.map((it, i) => {
    const next = items[i + 1];
    const nextIsSameDay = next && next.dayId === it.dayId;
    return {
      id: String(it.id),
      name: it.place.name,
      category: it.place.categoryName?.split('>').pop()?.trim() ?? '',
      rating: 0, reviewsCount: 0,                     // ⚠️ 장소 평점은 서버에 없음 (기록의 별점 평균을 쓰려면 4장)
      address: it.place.roadAddress ?? it.place.lotAddress ?? '',
      thumbnail: DEFAULT_PLACE_THUMB,                 // ⚠️ 장소 썸네일 없음
      lat: it.place.latitude, lng: it.place.longitude,
      order: it.visitOrder ?? 0,
      day: it.dayNumber ?? 0,                         // 0 = 날짜 미배정
      time: it.plannedArrivalTime?.slice(0, 5),
      note: it.memo ?? undefined,
      // "다음 장소로 이동" 정보: 같은 날 다음 핀의 transportMode 가 '이 핀에서 거기로 가는 방법'
      transitToNext: nextIsSameDay && next.transportMode
        ? { type: TRANSIT_TYPE[next.transportMode], label: TRANSIT_LABEL[next.transportMode], duration: '' }
        : undefined,
    };
  });
}
```

## 4. AI 추출 (`ExtractTab`, `handleExtractSubmit` 등)

현재: 2초 타이머 뒤 `INITIAL_EXTRACTED_PLACES`(고정)가 나옵니다. 입력은 URL 한 줄 또는 스크린샷 1장입니다.

### 4-1. 흐름

```
입력(링크/글/사진) → POST /ai-requests (202) → GET /ai-requests/{id} 반복 → SUCCESS
 → GET /place-candidates?aiRequestId=  (카카오 좌표가 붙은 후보)
 → 체크한 후보만 POST /place-candidates/add-to-trip
```

### 4-2. API

```ts
// src/api/extractApi.ts
import api from './axios';
import { waitForAi } from './aiApi';

export type CandidateDto = { id: number; rawName: string; rawAddress: string | null; categoryText: string | null;
  description: string | null; confidence: number; mapped: boolean; needsReview: boolean; selected: boolean; displayOrder: number;
  place: PlaceDto | null };

export type ExtractInput =
  | { kind: 'LINK'; url: string }
  | { kind: 'TEXT'; text: string }
  | { kind: 'IMAGE'; imageUrls: string[] };       // POST /files 로 먼저 올려서 받은 url (4-4)

export async function startExtraction(tripId: number | null, input: ExtractInput) {
  const body =
    input.kind === 'LINK' ? { inputType: 'LINK', sourceUrl: input.url }
    : input.kind === 'TEXT' ? { inputType: 'TEXT', sourceText: input.text }
    : { inputType: 'IMAGE', imageUrls: input.imageUrls };
  return (await api.post('/ai-requests', { tripId, ...body })).data as { id: number };
}

export const listCandidates = async (aiRequestId: number) =>
  (await api.get<CandidateDto[]>('/place-candidates', { params: { aiRequestId } })).data;

export const addCandidatesToTrip = async (tripId: number, candidateIds: number[]) =>
  (await api.post<{ added: ItemDto[]; skippedDuplicate: number[]; skippedNoCoordinate: number[] }>(
    '/place-candidates/add-to-trip', { tripId, candidateIds })).data;
```

### 4-3. `handleExtractSubmit` 교체

입력창 값이 `http`로 시작하면 `LINK`, 아니면 `TEXT`로 보내면 됩니다.

```ts
const handleExtractSubmit = async () => {
  if (!inputUrl.trim()) return;
  const tripId = Number(selectedTripForExtract) || null;     // 여행을 안 골랐으면 null 도 가능
  const signal = (extractSignal.current = { cancelled: false });
  setIsExtracting(true);
  try {
    const input: ExtractInput = /^https?:\/\//.test(inputUrl.trim())
      ? { kind: 'LINK', url: inputUrl.trim() }
      : { kind: 'TEXT', text: inputUrl.trim() };
    const { id } = await startExtraction(tripId, input);

    const result = await waitForAi(id, signal);
    if (!result) return showToast('시간이 오래 걸리고 있어요. 잠시 후 다시 확인해 주세요.');
    if (result.status === 'FAILED') return showToast('장소를 찾지 못했어요. 다른 링크로 다시 시도해 보세요.');

    await new Promise(r => setTimeout(r, 1500));             // 후보 저장(좌표 변환)이 SUCCESS 직후 이어서 진행됨
    const cands = await listCandidates(id);
    const places: ExtractedPlace[] = cands.map(toExtractedPlace);
    setExtractedPlaces(places);
    setSelectedExtractedPlaceIds(places.filter(p => p.extractionStatus === 'success').map(p => p.id)); // 확인 필요는 기본 선택 해제
    showToast(`AI가 장소 ${places.length}곳을 추출했습니다!`);
  } catch (e) {
    showToast(errorMessage(e));
  } finally {
    setIsExtracting(false);
  }
```

## 5. AI 맞춤 추천 (`handleAiTripRecommendation`)

현재는 입력창에 문장만 채우고 추출 탭으로 보냅니다. 실제 추천 API는 **여행을 기준**으로 합니다.


**프론트 입력값과 서버 값 맞추기**

| 프론트 (`recommendation*`) | 서버 | 변환 |
|---|---|---|
| `recommendationPeople` ('2' 등 인원 수) | `companionType` | 1명=`SOLO`, 2명=`COUPLE`, 3명 이상=`FAMILY` 또는 `FRIEND` (선택지를 직접 주는 편이 정확) |

```ts
const handleAiTripRecommendation = async () => {
```

- `accept`가 **422**면 AI가 말한 가게를 지도에서 못 찾은 것입니다(폐업·이름 변경). `message`에 가장 가까운 후보 이름이 있으니 토스트로 보여 주고, 3장의 장소 검색으로 직접 추가하게 안내하세요.
- 같은 여행에서 다시 요청하면 이전 추천·이미 있는 장소는 자동으로 제외됩니다.

- `PUT`은 **보낸 것만 남고 나머지는 해제**됩니다. 칩(체크박스) 화면의 **현재 선택 상태 전체**를 그대로 보내면 되고, 전부 해제하려면 빈 배열 `[]`을 보냅니다.
- 모르는 코드를 보내면 400이고 **아무것도 바뀌지 않습니다**(오류 메시지에 틀린 코드가 나옵니다). 선택지는 반드시 `listTravelStyles()` 결과의 `code`로 만드세요.
- 추천은 취향을 **추천 요청의 `styleCodes` → 이 여행의 취향 → 내 취향** 순서로 찾아 씁니다. 그래서 가입 직후나 프로필에서 "내 취향"을 한 번 저장해 두면, 추천 요청에 `styleCodes`를 안 보내도 그 취향이 적용됩니다.
- 여행을 만들 때 취향도 같이 보내는 기능은 없습니다. 여행을 만든 뒤 `saveTripStyles`를 따로 호출하세요.

```tsx
// 취향 선택 칩 (프로필 / 여행 만들기 / 추천 화면 공통)
const [styles, setStyles] = useState<StyleDto[]>([]);
const [picked, setPicked] = useState<string[]>([]);

useEffect(() => {
  Promise.all([listTravelStyles(), getMyStyles()]).then(([all, mine]) => {
    setStyles(all);
    setPicked(mine.map(s => s.code));
  });
}, []);

const toggle = (code: string) => setPicked(cur => (cur.includes(code) ? cur.filter(c => c !== code) : [...cur, code]));
const save = async () => { try { await saveMyStyles(picked); showToast('취향을 저장했어요.'); } catch (e) { showToast(errorMessage(e)); } };
```

## 11. 연동 순서 제안 (시연 효과 순)

1. **0장 + 1장** 로그인·토큰 → 이게 되어야 나머지가 됩니다.
2. **2장 + 3장** 여행 목록, 일정(일자·핀) 불러오기
3. **4장** AI 추출 (링크/글 → 후보 → 여행에 추가) ← 시연에서 가장 눈에 띕니다
4. **7장** 장소 기록 + 업로드
5. **8장** PinLog
6. **6장, 5장** 동선 최적화, 추천

화면 하나를 붙일 때마다: Swagger에서 먼저 같은 호출이 되는지 확인 → 코드 반영 → 막히면 **요청 주소, 보낸 body, 받은 응답(status + body)** 을 정훈에게 보내 주세요.
